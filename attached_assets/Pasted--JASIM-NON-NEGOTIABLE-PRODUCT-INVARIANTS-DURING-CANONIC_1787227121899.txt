# JASIM — NON-NEGOTIABLE PRODUCT INVARIANTS DURING CANONICAL MERGE
# PRESERVE BUBBLE-FIRST ARCHITECTURE + DIRECT RUNS + GENERATIVE SMART UI

أثناء دمج Runtime المطورة داخل مجلد `جاسم` الأصلي، ثبت القواعد التالية كـNON-NEGOTIABLE ARCHITECTURAL INVARIANTS.

لا تسمح لعملية الدمج أن تعيد جاسم إلى World-first أو Task-first أو Platform-first.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. THE PRODUCT IS BUBBLE-FIRST, NOT WORLD-FIRST
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

العلاقة الصحيحة:

Conversation
↓
Smart Generative Bubble
↓
optional durable World Runtime

وليس:

Conversation
↓
World
↓
UI

ولا:

World = separate product surface

World هي الحياة الداخلية الدائمة للفقاعة عندما تحتاج الفقاعة Persistence حقيقية.

User sees/interacts with Bubble.

World stores/runtime-manages durable internal state.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. WORLD MUST LIVE INSIDE / BEHIND A PERSISTENT BUBBLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

إذا Bubble تحتاج:

- entities
- workflows
- policies
- permissions
- versions
- transactions
- long-lived state

يمكن ربطها بـWorld.

لكن لا تعرض World للمستخدم كمنتج منفصل عن Bubble.

العلاقة:

SmartBubble {
  id
  conversationId?
  worldId? nullable
  ...
}

World {
  durable runtime state
}

مهم:
worldId OPTIONAL.

لا تجعل:
bubble.worldId NOT NULL
بشكل عام.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. SIMPLE BUBBLES DO NOT NEED WORLD
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Bubble قد تكون:

- result bubble
- comparison bubble
- interactive result
- temporary entity view
- ephemeral assistant surface

بدون World دائم.

اختبر:

"قارن لي هذه الخيارات"

Expected:
ephemeral/interactive bubble

NOT:
persistent world

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. RUN DOES NOT REQUIRE BUBBLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

هذه قاعدة أساسية.

Run يجب أن تستطيع أن تعمل بدون:

bubbleId
worldId

مثال:

"بع سيارتي"

Expected:
DIRECT_ACTION / DURABLE_RUN

وليس:
create persistent bubble

مثال:

"أعلن عن هذه الخدمة"

Expected:
ACTION / RUN

وليس:
Ads platform

مثال:

"راقب السعر"

Expected:
DURABLE_RUN

وليس:
World

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. BUBBLE DOES NOT REQUIRE RUN TO EXIST
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Bubble دائمة يمكن أن تستمر بعد انتهاء Run التي أنشأتها.

مثال:

Run A:
creates Bubble

تنتهي Run A.

Bubble تبقى.

لاحقًا:

Run B:
adds product

Run C:
changes policy

Run D:
schedules action

العلاقة:

Bubble
├── Run A
├── Run B
├── Run C
└── Run D

لا تجعل Bubble lifecycle مساوية لـRun lifecycle.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. TASK / RUN / BUBBLE / WORLD ARE DIFFERENT CONCEPTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

لا تدمجهم لمجرد تبسيط DB.

ثبت المعاني:

Task
= planned unit / orchestration work

Run
= durable execution lifecycle

Smart Bubble
= user-facing generative living surface

World
= optional durable runtime backing model for persistent Bubble

إذا implementation الأصلية تسمي الأشياء بشكل مختلف:
حافظ على semantics لا على الاسم.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
7. DO NOT FORCE PLATFORM CREATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

جاسم ليس Platform Builder فقط.

اختبر دائمًا:

"بع هذه السيارة"
→ Run

"أنشئ سوق سيارات"
→ Persistent Smart Bubble + optional World

"أعلن عن المنتج"
→ Action / Run

"أنشئ نظامًا لإدارة الحملات"
→ Persistent Smart Bubble

"ابحث لي عن موظف"
→ Workflow / Run

"أنشئ نظام توظيف للشركة"
→ Persistent Smart Bubble

إذا نفس المجال ينتج نفس الشكل دائمًا:
FAIL.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
8. OUTPUT ROUTER MUST REMAIN GENERAL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FINAL NON-NEGOTIABLE RULES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2. Bubble does not always need World.
3. Run does not need Bubble.
4. Direct actions do not require platforms.
5. Persistent Bubble survives individual Runs.
6. Bubbles are intelligent through central Jasim intelligence.

8. General Reference Resolution handles Bubble AND non-Bubble objects.
9. Web/Mobile render same Runtime truth.
10. HTML model changes presentation, not architecture.
11. No domain hardcoding.
12. No fake interactive Bubble.

14. Final architecture is Bubble-first, not World-first.
