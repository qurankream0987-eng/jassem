function getCol(hue, a) {
  let r,g,b;
  if (hue < 0.25) { const t=hue/0.25; r=0; g=Math.round(200+t*55); b=255; }
  else if (hue < 0.5) { const t=(hue-0.25)/0.25; r=Math.round(t*150); g=Math.round(255-t*155); b=255; }
  else if (hue < 0.75) { const t=(hue-0.5)/0.25; r=150+Math.round(t*105); g=100-Math.round(t*50); b=255-Math.round(t*100); }
  else { const t=(hue-0.75)/0.25; r=255-Math.round(t*255); g=50+Math.round(t*150); b=155+Math.round(t*100); }
  return `rgba(${r},${g},${b},${a})`;
}

    const pulse = Math.sin(sTime*0.2+n.phase)*0.3+0.7;
    const r = n.radius*pulse*sDpr;

    gr.addColorStop(0, getCol(n.hue, n.alpha*3));

  }

function sDrawNeural() {
  const cx = sW/2, cy = sH/2;
  for (const line of LINES) {
    const pts = []; const steps = 70;
    for (let i = 0; i <= steps; i++) {
      const t = i/steps;
      const dist = t*line.length*sDpr;
      const wave = Math.sin(t*line.freq*Math.PI*2+sTime*line.speed+line.phase)*line.amp*sDpr*(1-t*0.35);
      const noise = Math.sin(t*7+sTime+line.phase)*3.5*sDpr*(1-t);
      const x = cx+Math.cos(line.angle)*dist+Math.cos(line.angle+Math.PI/2)*(wave+noise);
      const y = cy+Math.sin(line.angle)*dist+Math.sin(line.angle+Math.PI/2)*(wave+noise);
      pts.push({x,y});
    }
    sCtx.strokeStyle = getCol(line.hue, line.alpha); sCtx.lineWidth = line.width;
    sCtx.beginPath(); sCtx.moveTo(pts[0].x, pts[0].y);
    for (let i = 1; i < pts.length; i++) sCtx.lineTo(pts[i].x, pts[i].y);
    sCtx.stroke();
    for (let i = 3; i < pts.length-3; i += 6) {
      sCtx.fillStyle = getCol(line.hue, line.alpha*0.45*(1-i/pts.length));
      sCtx.beginPath(); sCtx.arc(pts[i].x, pts[i].y, 1.1*sDpr, 0, Math.PI*2); sCtx.fill();

  }
}

function updateBubblePositions() {
  for (const b of BUBBLES) {
    b.px = b.x * bW; b.py = b.y * bH;
    b.pr = b.r * bDpr; b.baseR = b.r * bDpr;
  }
}

function hexToRgba(hex, alpha) {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return `rgba(${r}, ${g}, ${b}, ${alpha})`;
}

// ============================================================
// REALISTIC SOAP BUBBLE PHYSICS with pop animation
// ============================================================
function drawSoapBubble(b) {
  if (b.popped) {
    // Pop animation: shrink and fade
    const popProgress = Math.min((bTime - b.popTimer) / 0.5, 1);
    if (popProgress >= 1) return; // Fully popped

    const x = b.px, y = b.py;
    const shrink = 1 - popProgress;
    bCtx.save();
    bCtx.translate(x, y);
    bCtx.scale(shrink, shrink);
    bCtx.globalAlpha = 1 - popProgress;

    // Draw fading bubble
    bCtx.beginPath();
    bCtx.arc(0, 0, b.pr, 0, Math.PI * 2);
    bCtx.fillStyle = 'rgba(255,255,255,0.1)';
    bCtx.fill();
    bCtx.strokeStyle = hexToRgba(b.color, 0.3 * (1 - popProgress));
    bCtx.lineWidth = 2;
    bCtx.stroke();
    bCtx.restore();
    return;
  }

  const x = b.px, y = b.py, r = b.pr;
  const c1 = b.color, c2 = b.color2;

  // Advanced wobble physics
  const wobblePhase = bTime * 2.5 + b.id.charCodeAt(0) * 3;
  const wobblePhase2 = bTime * 1.8 + b.id.charCodeAt(0) * 5;
  const wobblePhase3 = bTime * 3.2 + b.id.charCodeAt(0) * 7;

  const scaleX = 1 + Math.sin(wobblePhase) * 0.04 + Math.cos(wobblePhase2) * 0.02;
  const scaleY = 1 + Math.cos(wobblePhase) * 0.04 + Math.sin(wobblePhase2) * 0.02;
  const skewX = Math.sin(wobblePhase3) * 0.03;
  const skewY = Math.cos(wobblePhase3) * 0.03;
  const rotation = Math.sin(wobblePhase * 0.5) * 0.05;
  const hoverScale = b.hover ? 1.08 : 1.0;

  bCtx.save();
  bCtx.translate(x, y);
  bCtx.rotate(rotation);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  bCtx.transform(1, skewX, skewY, 1, 0, 0);

  const radius = r;

  bCtx.beginPath();
  bCtx.arc(0, 0, radius, 0, Math.PI * 2);
  bCtx.clip();

  const img = BUBBLE_IMAGES[b.id];
  if (img && img.complete) {
    const imgSize = radius * 2.2;
    bCtx.globalAlpha = 0.55;
    bCtx.drawImage(img, -imgSize/2, -imgSize/2, imgSize, imgSize);
    bCtx.globalAlpha = 1;
  }

  bCtx.fillStyle = 'rgba(0, 0, 0, 0.35)';
  bCtx.fillRect(-radius, -radius, radius * 2, radius * 2);

  const filmGrad = bCtx.createRadialGradient(-radius*0.3, -radius*0.3, 0, 0, 0, radius * 1.1);
  filmGrad.addColorStop(0, 'rgba(255,255,255,0.15)');
  filmGrad.addColorStop(0.1, hexToRgba(c1, 0.12));
  filmGrad.addColorStop(0.25, hexToRgba(c2, 0.08));
  filmGrad.addColorStop(0.4, hexToRgba(c1, 0.06));
  filmGrad.addColorStop(0.6, hexToRgba(c2, 0.04));
  filmGrad.addColorStop(0.8, hexToRgba(c1, 0.03));
  filmGrad.addColorStop(0.95, 'rgba(255,255,255,0.02)');
  filmGrad.addColorStop(1, 'rgba(0,0,0,0.01)');
  bCtx.fillStyle = filmGrad;
  bCtx.fillRect(-radius, -radius, radius * 2, radius * 2);
  bCtx.restore();

  // Fresnel rim
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  const fresnelGrad = bCtx.createRadialGradient(0, 0, radius * 0.7, 0, 0, radius * 1.05);
  fresnelGrad.addColorStop(0, 'transparent');
  fresnelGrad.addColorStop(0.6, 'rgba(255,255,255,0.02)');
  fresnelGrad.addColorStop(0.85, hexToRgba(c2, 0.15));
  fresnelGrad.addColorStop(0.95, hexToRgba(c1, 0.25));
  fresnelGrad.addColorStop(1, 'rgba(255,255,255,0.35)');
  bCtx.strokeStyle = fresnelGrad;
  bCtx.lineWidth = 2.5;
  bCtx.beginPath();
  bCtx.arc(0, 0, radius, 0, Math.PI * 2);
  bCtx.stroke();
  bCtx.restore();

  // Chromatic aberration
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  bCtx.globalCompositeOperation = 'screen';
  const chromaOffset = 2 * bDpr;
  bCtx.strokeStyle = 'rgba(255, 80, 80, 0.12)';
  bCtx.lineWidth = 3;
  bCtx.beginPath();
  bCtx.arc(-chromaOffset, -chromaOffset, radius, 0, Math.PI * 2);
  bCtx.stroke();
  bCtx.strokeStyle = 'rgba(0, 200, 255, 0.12)';
  bCtx.beginPath();
  bCtx.arc(chromaOffset, chromaOffset, radius, 0, Math.PI * 2);
  bCtx.stroke();
  bCtx.globalCompositeOperation = 'source-over';
  bCtx.restore();

  drawCaustics(x, y, radius * Math.max(scaleX, scaleY) * hoverScale, b.color, bTime);

  // Specular highlight
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  const hlX = -radius * 0.35 + Math.sin(bTime * 1.5) * radius * 0.08;
  const hlY = -radius * 0.35 + Math.cos(bTime * 1.2) * radius * 0.08;
  const hlGrad = bCtx.createRadialGradient(hlX, hlY, 0, hlX, hlY, radius * 0.4);
  hlGrad.addColorStop(0, 'rgba(255,255,255,0.8)');
  hlGrad.addColorStop(0.2, 'rgba(255,255,255,0.4)');
  hlGrad.addColorStop(0.5, 'rgba(255,255,255,0.1)');
  hlGrad.addColorStop(1, 'transparent');
  bCtx.fillStyle = hlGrad;
  bCtx.beginPath();
  bCtx.arc(hlX, hlY, radius * 0.4, 0, Math.PI * 2);
  bCtx.fill();
  bCtx.restore();

  // Secondary caustic
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  const causticX = radius * 0.25;
  const causticY = radius * 0.3;
  const causticGrad = bCtx.createRadialGradient(causticX, causticY, 0, causticX, causticY, radius * 0.3);
  causticGrad.addColorStop(0, hexToRgba(c2, 0.2));
  causticGrad.addColorStop(0.4, hexToRgba(c1, 0.12));
  causticGrad.addColorStop(1, 'transparent');
  bCtx.fillStyle = causticGrad;
  bCtx.beginPath();
  bCtx.arc(causticX, causticY, radius * 0.3, 0, Math.PI * 2);
  bCtx.fill();
  bCtx.restore();

  // Glow
  const glowGrad = bCtx.createRadialGradient(x, y, radius * 0.6, x, y, radius * 2.2);
  glowGrad.addColorStop(0, hexToRgba(b.color, 0.06));
  glowGrad.addColorStop(0.5, hexToRgba(b.color2, 0.03));
  glowGrad.addColorStop(1, 'transparent');
  bCtx.fillStyle = glowGrad;
  bCtx.beginPath();
  bCtx.arc(x, y, radius * 2.2, 0, Math.PI * 2);
  bCtx.fill();

  // Label
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  bCtx.fillStyle = 'rgba(255,255,255,0.95)';
  bCtx.font = `700 ${13 * bDpr}px 'Noto Sans Arabic'`;
  bCtx.textAlign = 'center';
  bCtx.textBaseline = 'middle';
  bCtx.shadowColor = 'rgba(0,0,0,0.8)';
  bCtx.shadowBlur = 8 * bDpr;
  bCtx.fillText(b.label, 0, radius * 0.05);
  bCtx.shadowBlur = 0;
  bCtx.restore();

  // Status dot
  bCtx.save();
  bCtx.translate(x, y);
  bCtx.scale(scaleX * hoverScale, scaleY * hoverScale);
  bCtx.fillStyle = hexToRgba(b.color, 0.9);
  bCtx.beginPath();
  bCtx.arc(0, -radius * 0.3, 4 * bDpr, 0, Math.PI * 2);
  bCtx.fill();
  bCtx.restore();
}

function drawCaustics(cx, cy, radius, color, t) {
  bCtx.save();
  bCtx.translate(cx, cy);
  bCtx.globalCompositeOperation = 'screen';
  const causticCount = 12;
  for (let i = 0; i < causticCount; i++) {
    const angle = (i / causticCount) * Math.PI * 2 + t * 0.3 + i * 1.5;
    const dist = 30 + Math.sin(t * 0.8 + i * 2) * 15;
    const causticX = Math.cos(angle) * dist;
    const causticY = Math.sin(angle) * dist;
    const causticSize = 8 + Math.sin(t * 1.2 + i) * 4;
    const alpha = 0.03 + Math.sin(t * 0.5 + i) * 0.02;
    const grad = bCtx.createRadialGradient(causticX, causticY, 0, causticX, causticY, causticSize);
    grad.addColorStop(0, hexToRgba(color, alpha * 2));
    grad.addColorStop(0.5, hexToRgba(color, alpha));
    grad.addColorStop(1, 'transparent');
    bCtx.fillStyle = grad;
    bCtx.beginPath();
    bCtx.arc(causticX, causticY, causticSize, 0, Math.PI * 2);
    bCtx.fill();
  }
  bCtx.globalCompositeOperation = 'source-over';
  bCtx.restore();
}

function createPopParticles(x, y, color) {
  for (let i = 0; i < 40; i++) {
    const p = document.createElement('div');
    p.className = 'pop-particle';
    const size = 2 + Math.random() * 14;
    const angle = (Math.PI * 2 * i) / 40 + Math.random() * 0.5;
    const dist = 50 + Math.random() * 180;
    const tx = Math.cos(angle) * dist;
    const ty = Math.sin(angle) * dist;
    const duration = 0.5 + Math.random() * 0.7;
    p.style.cssText = `left: ${x}px; top: ${y}px; width: ${size}px; height: ${size}px; background: radial-gradient(circle, ${color}ff, ${color}80, transparent); box-shadow: 0 0 ${size*2}px ${color}80, 0 0 ${size*4}px ${color}40, inset 0 0 ${size}px rgba(255,255,255,0.5); --tx: ${tx}px; --ty: ${ty}px; animation-duration: ${duration}s;`;
    document.body.appendChild(p);
    setTimeout(() => p.remove(), duration * 1000);
  }
  for (let i = 0; i < 20; i++) {
    const p = document.createElement('div');
    p.className = 'pop-particle';
    const size = 1 + Math.random() * 4;
    const angle = Math.random() * Math.PI * 2;
    const dist = 100 + Math.random() * 250;
    const tx = Math.cos(angle) * dist;
    const ty = Math.sin(angle) * dist;
    p.style.cssText = `left: ${x}px; top: ${y}px; width: ${size}px; height: ${size}px; background: white; box-shadow: 0 0 ${size*4}px white, 0 0 ${size*8}px ${color}60; --tx: ${tx}px; --ty: ${ty}px; animation-duration: 0.6s;`;
    document.body.appendChild(p);
    setTimeout(() => p.remove(), 600);
  }
}

function getBubbleAt(mx, my) {
  for (const b of BUBBLES) {
    if (b.popped) continue;
    const dx = mx * bDpr - b.px;
    const dy = my * bDpr - b.py;
    if (Math.hypot(dx, dy) < b.pr * 1.15) return b;
  }
  return null;
}

// BUBBLE INTERACTION — Long press to pop, short press to open

function getCanvasCoords(e) {
  const rect = bCanvas.getBoundingClientRect();
  return { mx: e.clientX - rect.left, my: e.clientY - rect.top, cx: e.clientX, cy: e.clientY };
}
function getTouchCanvasCoords(touch) {
  const rect = bCanvas.getBoundingClientRect();
  return { mx: touch.clientX - rect.left, my: touch.clientY - rect.top, cx: touch.clientX, cy: touch.clientY };
}

let pressTimer = null;
let pressStartTime = 0;
let isLongPress = false;
const LONG_PRESS_DURATION = 400;
const bubbleHint = document.getElementById('bubbleHint');

function showBubbleHint(x, y) {
  bubbleHint.style.left = (x + 20) + 'px';
  bubbleHint.style.top = (y - 30) + 'px';
  bubbleHint.classList.add('visible');
}
function hideBubbleHint() {
  bubbleHint.classList.remove('visible');
}

function popBubble(b, screenX, screenY) {
  if (b.popped) return;
  b.popped = true;
  b.popTimer = bTime;
  createPopParticles(screenX, screenY, b.color);

  // Respawn after 3 seconds
  setTimeout(() => {
    b.popped = false;
    b.popTimer = 0;
  }, 3000);
}

document.addEventListener('mousemove', (e) => {
  const { mx, my } = getCanvasCoords(e);
  bMouse.x = mx; bMouse.y = my;
  if (draggedBubble && bMouse.down) {
    const newPx = mx * bDpr - dragOffset.x;
    const newPy = my * bDpr - dragOffset.y;
    draggedBubble.dragVx = (newPx - draggedBubble.px) * 0.3;
    draggedBubble.dragVy = (newPy - draggedBubble.py) * 0.3;
    draggedBubble.px = newPx; draggedBubble.py = newPy;
    document.body.style.cursor = 'grabbing';
    return;
  }
  const b = getBubbleAt(mx, my);
  BUBBLES.forEach(bb => bb.hover = false);
  if (b) { 
    b.hover = true; 
    document.body.style.cursor = 'grab'; 
  }
  else { 
    document.body.style.cursor = 'default'; 
    hideBubbleHint();
  }
});

document.addEventListener('mousedown', (e) => {
  if (e.target.closest('.soap-window')) return;
  const { mx, my, cx, cy } = getCanvasCoords(e);
  const b = getBubbleAt(mx, my);
  if (b) {
    bMouse.down = true;
    draggedBubble = b;
    dragOffset.x = mx * bDpr - b.px;
    dragOffset.y = my * bDpr - b.py;
    b.dragVx = 0; b.dragVy = 0;

    // Long press detection
    isLongPress = false;
    pressStartTime = performance.now();
    pressTimer = setTimeout(() => {
      isLongPress = true;
      showBubbleHint(cx, cy);
      // Visual feedback for long press
      b.hover = true;
    }, LONG_PRESS_DURATION);
  }
});

document.addEventListener('mouseup', (e) => {
  if (pressTimer) {
    clearTimeout(pressTimer);
    pressTimer = null;
  }
  hideBubbleHint();

  if (!draggedBubble) return;
  const { mx, my, cx, cy } = getCanvasCoords(e);
  const dx = mx * bDpr - draggedBubble.px;
  const dy = my * bDpr - draggedBubble.py;

  const pressDuration = performance.now() - pressStartTime;

  if (isLongPress || pressDuration > LONG_PRESS_DURATION) {
    // Long press → POP the bubble!
    popBubble(draggedBubble, cx, cy);
  } else if (Math.hypot(dx - dragOffset.x, dy - dragOffset.y) < 20 * bDpr) {
    // Short press + no drag → Open window
    if (!draggedBubble.popped) {
      expandBubble(draggedBubble, cx, cy);
    }
  } else {
    draggedBubble.vx = draggedBubble.dragVx;
    draggedBubble.vy = draggedBubble.dragVy;
  }

  draggedBubble = null;
  bMouse.down = false;
  isLongPress = false;
  document.body.style.cursor = 'default';
});

document.addEventListener('touchstart', (e) => {
  if (e.target.closest('.soap-window')) return;
  const touch = e.touches[0];
  const { mx, my } = getTouchCanvasCoords(touch);
  const b = getBubbleAt(mx, my);
  if (b) {
    bMouse.down = true;
    draggedBubble = b;
    dragOffset.x = mx * bDpr - b.px;
    dragOffset.y = my * bDpr - b.py;

    isLongPress = false;
    pressStartTime = performance.now();
    pressTimer = setTimeout(() => {
      isLongPress = true;
      showBubbleHint(touch.clientX, touch.clientY);
    }, LONG_PRESS_DURATION);

    e.preventDefault();
  }
}, { passive: false });

document.addEventListener('touchmove', (e) => {
  if (!draggedBubble) return;
  const touch = e.touches[0];
  const { mx, my } = getTouchCanvasCoords(touch);
  const newPx = mx * bDpr - dragOffset.x;
  const newPy = my * bDpr - dragOffset.y;
  draggedBubble.dragVx = (newPx - draggedBubble.px) * 0.3;
  draggedBubble.dragVy = (newPy - draggedBubble.py) * 0.3;
  draggedBubble.px = newPx; draggedBubble.py = newPy;

  // Cancel long press if dragged
  if (Math.hypot(newPx - draggedBubble.px, newPy - draggedBubble.py) > 10 * bDpr) {
    if (pressTimer) { clearTimeout(pressTimer); pressTimer = null; }
    hideBubbleHint();
  }

  e.preventDefault();
}, { passive: false });

document.addEventListener('touchend', (e) => {
  if (pressTimer) {
    clearTimeout(pressTimer);
    pressTimer = null;
  }
  hideBubbleHint();

  if (!draggedBubble) return;
  const touch = e.changedTouches[0];
  const { mx, my, cx, cy } = getTouchCanvasCoords(touch);
  const dx = mx * bDpr - draggedBubble.px;
  const dy = my * bDpr - draggedBubble.py;

  const pressDuration = performance.now() - pressStartTime;

  if (isLongPress || pressDuration > LONG_PRESS_DURATION) {
    popBubble(draggedBubble, cx, cy);
  } else if (Math.hypot(dx - dragOffset.x, dy - dragOffset.y) < 20 * bDpr) {
    if (!draggedBubble.popped) {
      expandBubble(draggedBubble, cx, cy);
    }

    draggedBubble.vy = draggedBubble.dragVy;
  }

  draggedBubble = null;
  bMouse.down = false;
  isLongPress = false;
});

function updateWindowDepth() {
  const sorted = Array.from(openWindows.entries())
    .filter(([_, d]) => d.state !== 'minimized')
    .sort((a, b) => parseInt(a[1].element.style.zIndex) - parseInt(b[1].element.style.zIndex));
  sorted.forEach(([_, data], i) => {
    const depth = sorted.length - i;
    data.element.classList.remove('depth-1', 'depth-2', 'depth-3');
    if (depth <= 3) data.element.classList.add(`depth-${depth}`);
  });
}

function minimizeWindow(id) {
  const winData = openWindows.get(id);
  if (!winData || winData.state === 'minimized') return;

  const win = winData.element;
  winData.prevState = winData.state;
  winData.state = 'minimized';

  const rect = win.getBoundingClientRect();
  winData.normalBounds = {
    left: rect.left, top: rect.top,
    width: rect.width, height: rect.height
  };

  win.classList.add('animating-minimize');

  setTimeout(() => {
    win.classList.remove('state-normal', 'state-maximized', 'state-snapped-left', 'state-snapped-right');
    win.classList.add('state-minimized');
    win.classList.remove('animating-minimize');
    // Window is now hidden (opacity:0, scale:0) — only dock shows
    addToDock(id);
    updateWindowDepth();
  }, 500);
}

function maximizeWindow(id) {
  const winData = openWindows.get(id);
  if (!winData) return;

  const win = winData.element;

  if (winData.state === 'maximized') {
    restoreWindow(id);
    return;
  }

  if (winData.state !== 'minimized') {
    const rect = win.getBoundingClientRect();
    winData.normalBounds = {
      left: rect.left, top: rect.top,
      width: rect.width, height: rect.height
    };
  }

  winData.prevState = winData.state;
  winData.state = 'maximized';

  win.classList.add('animating-maximize');

  setTimeout(() => {
    win.classList.remove('state-normal', 'state-minimized', 'state-snapped-left', 'state-snapped-right');
    win.classList.add('state-maximized');
    win.classList.remove('animating-maximize');
    removeFromDock(id);
    updateWindowDepth();
  }, 350);
}

function restoreWindow(id) {
  const winData = openWindows.get(id);
  if (!winData) return;

  const win = winData.element;
  const bounds = winData.normalBounds;

  winData.state = winData.prevState || 'normal';
  winData.prevState = null;

  win.classList.add('animating-restore');

  setTimeout(() => {
    win.classList.remove('state-maximized', 'state-minimized', 'state-snapped-left', 'state-snapped-right');
    win.classList.add('state-normal');
    win.classList.remove('animating-restore');

    win.style.left = bounds.left + 'px';
    win.style.top = bounds.top + 'px';
    win.style.width = bounds.width + 'px';
    win.style.height = bounds.height + 'px';

    removeFromDock(id);
    updateWindowDepth();
  }, 400);
}

function closeWindow(id) {
  const winData = openWindows.get(id);
  if (!winData) return;

  const win = winData.element;
  win.classList.add('animating-minimize');

  setTimeout(() => {
    win.remove();
    openWindows.delete(id);
    removeFromDock(id);
    updateWindowDepth();
  }, 500);
}

function setupResizeHandles(win, id) {
  const handles = win.querySelectorAll('.resize-handle');
  handles.forEach(handle => {
    handle.addEventListener('mousedown', (e) => {
      e.stopPropagation();
      startResize(e, win, id, handle.dataset.dir);
    });
    handle.addEventListener('touchstart', (e) => {
      e.stopPropagation();
      startResize(e.touches[0], win, id, handle.dataset.dir);
    }, { passive: false });
  });
}

let resizingWindow = null;
let resizeData = null;

function startResize(e, win, id, dir) {
  if (openWindows.get(id).state === 'maximized') return;

  resizingWindow = win;
  const rect = win.getBoundingClientRect();
  resizeData = {
    dir: dir,
    startX: e.clientX,
    startY: e.clientY,
    startLeft: rect.left,
    startTop: rect.top,
    startWidth: rect.width,
    startHeight: rect.height,
    minWidth: 280,
    minHeight: 180
  };

  document.body.style.cursor = getResizeCursor(dir);
  e.preventDefault && e.preventDefault();
}

function getResizeCursor(dir) {
  const cursors = {
    n: 'ns-resize', s: 'ns-resize', e: 'ew-resize', w: 'ew-resize',
    ne: 'nesw-resize', nw: 'nwse-resize', se: 'nwse-resize', sw: 'nesw-resize'
  };
  return cursors[dir] || 'default';
}

function handleResizeMove(e) {
  if (!resizingWindow || !resizeData) return;

  const clientX = e.touches ? e.touches[0].clientX : e.clientX;
  const clientY = e.touches ? e.touches[0].clientY : e.clientY;

  const dx = clientX - resizeData.startX;
  const dy = clientY - resizeData.startY;
  const dir = resizeData.dir;

  let newLeft = resizeData.startLeft;
  let newTop = resizeData.startTop;
  let newWidth = resizeData.startWidth;
  let newHeight = resizeData.startHeight;

  if (dir.includes('e')) newWidth = Math.max(resizeData.minWidth, resizeData.startWidth + dx);
  if (dir.includes('w')) {
    const proposedWidth = resizeData.startWidth - dx;
    if (proposedWidth >= resizeData.minWidth) {
      newWidth = proposedWidth;
      newLeft = resizeData.startLeft + dx;
    }
  }
  if (dir.includes('s')) newHeight = Math.max(resizeData.minHeight, resizeData.startHeight + dy);
  if (dir.includes('n')) {
    const proposedHeight = resizeData.startHeight - dy;
    if (proposedHeight >= resizeData.minHeight) {
      newHeight = proposedHeight;
      newTop = resizeData.startTop + dy;
    }
  }

  resizingWindow.style.left = newLeft + 'px';
  resizingWindow.style.top = newTop + 'px';
  resizingWindow.style.width = newWidth + 'px';
  resizingWindow.style.height = newHeight + 'px';

  const winData = openWindows.get(resizingWindow.id.replace('bw-', ''));
  if (winData) {
    winData.normalBounds = { left: newLeft, top: newTop, width: newWidth, height: newHeight };
  }
}

document.addEventListener('mousemove', handleResizeMove);
document.addEventListener('touchmove', handleResizeMove, { passive: false });

document.addEventListener('mouseup', () => {
  if (resizingWindow) {
    resizingWindow = null;
    resizeData = null;
    document.body.style.cursor = 'default';
  }
});
document.addEventListener('touchend', () => {
  if (resizingWindow) {
    resizingWindow = null;
    resizeData = null;
    document.body.style.cursor = 'default';
  }
});

function checkSnapZone(x, y, width, height) {
  const snapThreshold = 80;
  const screenW = window.innerWidth;
  const screenH = window.innerHeight;

  if (x < snapThreshold && y > snapThreshold && y + height < screenH - snapThreshold) {
    return 'left';
  }
  if (x + width > screenW - snapThreshold && y > snapThreshold && y + height < screenH - snapThreshold) {
    return 'right';
  }
  if (y < snapThreshold && x > snapThreshold && x + width < screenW - snapThreshold) {
    return 'maximize';
  }
  return null;
}

function showSnapPreview(zone) {
  const preview = document.getElementById('snapPreview');
  preview.style.display = 'block';

  if (zone === 'left') {
    preview.style.left = '8px'; preview.style.top = '8px';
    preview.style.width = 'calc(50vw - 16px)'; preview.style.height = 'calc(100vh - 16px)';
    magLeft.classList.add('active');
  } else if (zone === 'right') {
    preview.style.left = 'calc(50vw + 8px)'; preview.style.top = '8px';
    preview.style.width = 'calc(50vw - 16px)'; preview.style.height = 'calc(100vh - 16px)';
    magRight.classList.add('active');
  } else if (zone === 'maximize') {
    preview.style.left = '8px'; preview.style.top = '8px';
    preview.style.width = 'calc(100vw - 16px)'; preview.style.height = 'calc(100vh - 16px)';
    magTop.classList.add('active');
  }
}

function hideSnapPreview() {
  document.getElementById('snapPreview').style.display = 'none';
  magLeft.classList.remove('active');
  magRight.classList.remove('active');
  magTop.classList.remove('active');
}

function applySnap(id, zone) {
  const winData = openWindows.get(id);
  if (!winData) return;

  const win = winData.element;
  const rect = win.getBoundingClientRect();
  winData.normalBounds = {
    left: rect.left, top: rect.top,
    width: rect.width, height: rect.height
  };

  winData.prevState = winData.state;
  win.classList.add('snapping');

  if (zone === 'left') {
    winData.state = 'snapped-left';
    win.classList.remove('state-normal', 'state-maximized', 'state-minimized', 'state-snapped-right');
    win.classList.add('state-snapped-left');
  } else if (zone === 'right') {
    winData.state = 'snapped-right';
    win.classList.remove('state-normal', 'state-maximized', 'state-minimized', 'state-snapped-left');
    win.classList.add('state-snapped-right');
  } else if (zone === 'maximize') {
    maximizeWindow(id);
    win.classList.remove('snapping');
    return;
  }

  hideSnapPreview();
  setTimeout(() => win.classList.remove('snapping'), 350);
}

function makeWindowDraggable(win, bid) {
  const header = win.querySelector('.sw-header');
  let isDragging = false;
  let startX, startY, startLeft, startTop;
  let dragMoved = false;
  let velocityX = 0, velocityY = 0;
  let lastX, lastY, lastTime;

  header.addEventListener('mousedown', (e) => {
    if (e.target.closest('.sw-btn')) return;

    const winData = openWindows.get(bid);
    if (winData.state === 'maximized') {
      const normalW = winData.normalBounds.width;
      const normalH = winData.normalBounds.height;
      winData.normalBounds.left = e.clientX - normalW / 2;
      winData.normalBounds.top = e.clientY - 20;
      restoreWindow(bid);
      isDragging = true;
      startX = e.clientX; startY = e.clientY;
      startLeft = winData.normalBounds.left;
      startTop = winData.normalBounds.top;
    } else {
      isDragging = true;
      startX = e.clientX; startY = e.clientY;
      const rect = win.getBoundingClientRect();
      startLeft = rect.left; startTop = rect.top;
    }

    lastX = e.clientX; lastY = e.clientY; lastTime = performance.now();
    win.classList.add('dragging');
    win.style.zIndex = windowZIndex++;
    updateWindowDepth();
    dragMoved = false;
    e.preventDefault();
  });

  document.addEventListener('mousemove', (e) => {
    if (!isDragging) return;
    dragMoved = true;

    const now = performance.now();
    const dt = now - lastTime;
    if (dt > 0) {
      velocityX = (e.clientX - lastX) / dt * 16;
      velocityY = (e.clientY - lastY) / dt * 16;
    }
    lastX = e.clientX; lastY = e.clientY; lastTime = now;

    win.style.left = (startLeft + e.clientX - startX) + 'px';
    win.style.top = (startTop + e.clientY - startY) + 'px';
    win.style.transform = 'none';

    const rect = win.getBoundingClientRect();
    const snapZone = checkSnapZone(rect.left, rect.top, rect.width, rect.height);
    if (snapZone) {
      showSnapPreview(snapZone);
      snapTarget = snapZone;
    } else {
      hideSnapPreview();
      snapTarget = null;
    }
  });

  document.addEventListener('mouseup', (e) => {
    if (!isDragging) return;
    isDragging = false;
    win.classList.remove('dragging');

    if (snapTarget) {
      applySnap(bid, snapTarget);
      snapTarget = null;
      return;
    }

    const rect = win.getBoundingClientRect();
    const screenW = window.innerWidth;
    const screenH = window.innerHeight;
    let finalLeft = rect.left;
    let finalTop = rect.top;
    let bounced = false;

    if (rect.left < 10) { finalLeft = 10; bounced = true; }
    if (rect.left + rect.width > screenW - 10) { finalLeft = screenW - rect.width - 10; bounced = true; }
    if (rect.top < 10) { finalTop = 10; bounced = true; }
    if (rect.top + rect.height > screenH - 10) { finalTop = screenH - rect.height - 10; bounced = true; }

    if (bounced) {
      win.classList.add('snapping');
      win.style.transition = 'left 0.4s var(--spring), top 0.4s var(--spring)';
      win.style.left = finalLeft + 'px';
      win.style.top = finalTop + 'px';
      setTimeout(() => {
        win.style.transition = '';
        win.classList.remove('snapping');
      }, 400);
    }

    const winData = openWindows.get(bid);
    if (winData && winData.state === 'normal') {
      winData.normalBounds = {
        left: parseFloat(win.style.left) || finalLeft,
        top: parseFloat(win.style.top) || finalTop,
        width: rect.width, height: rect.height
      };
    }
  });

  header.addEventListener('touchstart', (e) => {
    if (e.target.closest('.sw-btn')) return;
    const touch = e.touches[0];
    const winData = openWindows.get(bid);
    if (winData.state === 'maximized') {
      const normalW = winData.normalBounds.width;
      winData.normalBounds.left = touch.clientX - normalW / 2;
      winData.normalBounds.top = touch.clientY - 20;
      restoreWindow(bid);
    }
    isDragging = true;
    startX = touch.clientX; startY = touch.clientY;
    const rect = win.getBoundingClientRect();
    startLeft = rect.left; startTop = rect.top;
    lastX = touch.clientX; lastY = touch.clientY; lastTime = performance.now();
    win.classList.add('dragging');
    win.style.zIndex = windowZIndex++;
  }, { passive: false });

  document.addEventListener('touchmove', (e) => {
    if (!isDragging) return;
    const touch = e.touches[0];
    const now = performance.now();
    const dt = now - lastTime;
    if (dt > 0) {
      velocityX = (touch.clientX - lastX) / dt * 16;
      velocityY = (touch.clientY - lastY) / dt * 16;
    }
    lastX = touch.clientX; lastY = touch.clientY; lastTime = now;

    win.style.left = (startLeft + touch.clientX - startX) + 'px';
    win.style.top = (startTop + touch.clientY - startY) + 'px';
    win.style.transform = 'none';

    const rect = win.getBoundingClientRect();
    const snapZone = checkSnapZone(rect.left, rect.top, rect.width, rect.height);
    if (snapZone) { showSnapPreview(snapZone); snapTarget = snapZone; }
    else { hideSnapPreview(); snapTarget = null; }
  }, { passive: false });

  document.addEventListener('touchend', (e) => {
    if (!isDragging) return;
    isDragging = false;
    win.classList.remove('dragging');
    if (snapTarget) {
      applySnap(bid, snapTarget);
      snapTarget = null;
      hideSnapPreview();
    }
  });
}

function bAnimate() {
  requestAnimationFrame(bAnimate);
  const now = performance.now();
  const dt = Math.min((now - bLast) / 1000, 0.05); bLast = now; bTime += dt;
  bCtx.clearRect(0, 0, bW, bH);

  for (const b of BUBBLES) {
    if (draggedBubble === b) continue;
    if (b.popped) {
      // Still animate popped bubbles (fade out)
      const popProgress = (bTime - b.popTimer) / 0.5;
      if (popProgress < 1) {
        // Keep position but let it drift slightly
        b.px += b.vx * 0.1;
        b.py += b.vy * 0.1;
      }
      continue; // drawSoapBubble handles the pop animation
    }
    b.px += b.vx; b.py += b.vy;
    b.vx *= 0.95; b.vy *= 0.95;
    const floatX = Math.sin(bTime * 0.7 + b.id.charCodeAt(0) * 2) * 18 * bDpr;
    const floatY = Math.cos(bTime * 0.5 + b.id.charCodeAt(0) * 3) * 12 * bDpr;
    b.px += floatX * 0.01; b.py += floatY * 0.01;
    const margin = b.pr * 1.5;
    if (b.px < margin) { b.px = margin; b.vx *= -0.5; }
    if (b.px > bW - margin) { b.px = bW - margin; b.vx *= -0.5; }
    if (b.py < margin) { b.py = margin; b.vy *= -0.5; }
    if (b.py > bH - margin) { b.py = bH - margin; b.vy *= -0.5; }
  }

  for (let i = 0; i < BUBBLES.length; i++) {
    for (let j = i + 1; j < BUBBLES.length; j++) {
      const dx = BUBBLES[j].px - BUBBLES[i].px;
      const dy = BUBBLES[j].py - BUBBLES[i].py;
      const d = Math.hypot(dx, dy);
      if (d < 400 * bDpr) {
        const alpha = (1 - d / (400 * bDpr)) * 0.08;
        const steps = 20;
        bCtx.strokeStyle = `rgba(100,180,255,${alpha})`;
        bCtx.lineWidth = 0.8;
        bCtx.beginPath();
        for (let s = 0; s <= steps; s++) {
          const t = s / steps;
          const px = BUBBLES[i].px + dx * t;
          const py = BUBBLES[i].py + dy * t;
          const wave = Math.sin(t * Math.PI * 4 + bTime * 2 + i) * 5 * bDpr * (1 - Math.abs(t - 0.5) * 2);
          const nx = -dy / d;
          const ny = dx / d;
          if (s === 0) bCtx.moveTo(px + nx * wave, py + ny * wave);
          else bCtx.lineTo(px + nx * wave, py + ny * wave);
        }
        bCtx.stroke();
      }
    }
  }

  for (const b of BUBBLES) {
    drawSoapBubble(b);
  }
}
bAnimate();

function autoResize(el) { el.style.height = 'auto'; el.style.height = Math.min(el.scrollHeight, 120) + 'px'; }
function scrollToBottom() { chatArea.scrollTo({ top: chatArea.scrollHeight, behavior: 'smooth' }); scrollBtn.classList.remove('visible'); }

  const nearBottom = chatArea.scrollHeight - chatArea.scrollTop - chatArea.clientHeight < 100;
  scrollBtn.classList.toggle('visible', !nearBottom);
});

function sendMessage() {
  const text = messageInput.value.trim();
  if (!text) return;
  welcomeOverlay.classList.add('hidden');
  addMessage(text, 'user');
  messageInput.value = ''; messageInput.style.height = 'auto';
  typingIndicator.classList.remove('hidden'); scrollToBottom();
  setTimeout(() => { typingIndicator.classList.add('hidden'); streamAIResponse(); }, 1800);
}

let isStreaming = false, streamInterval;
function streamAIResponse() {
  isStreaming = true;
  const responses = [
    "ممتاز! دعني أُعدّ لك خطة تنفيذية مفصلة...",
    "أولاً: نحتاج لتحديد نموذج الأعمال. هل تريد B2B2C أم B2C فقط؟",
    "ثانياً: البنية التقنية. أنصحك بـ FastAPI للـ Backend وFlutter للواجة."
  ];
  const response = responses[Math.floor(Math.random()*responses.length)];
  const words = response.split(' ');
  const group = document.createElement('div');
  group.className = 'message-group ai';
  group.innerHTML = `<div class="message-bubble"><span id="liveText"></span></div><div class="message-controls" style="opacity:0.3"><button class="msg-btn" title="إيقاف" onclick="stopStream()"><svg width="14" height="14" viewBox="0 0 24 24" fill="currentColor"><rect x="6" y="6" width="12" height="12" rx="2"/></svg></button></div>`;
  chatArea.insertBefore(group, typingIndicator); scrollToBottom();
  const liveText = group.querySelector('#liveText');
  let i = 0;
  streamInterval = setInterval(() => {
    if (i >= words.length) { stopStream(); return; }
    liveText.textContent += (i > 0 ? ' ' : '') + words[i]; scrollToBottom(); i++;
  }, 90);
}
function stopStream() {
  if (!isStreaming) return;
  clearInterval(streamInterval); isStreaming = false;
}
